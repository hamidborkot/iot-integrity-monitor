        digest.update(
            json.dumps(canonical, separators=(",", ":")).encode("utf-8")
        )
        digest.update(b"\n")
    return digest.hexdigest()

def scan_inputs(assets, scan_kinds, out):
    errors = []
    archives = []

    for root in BACKUPS + [WORK]:
        if not root.is_dir():
            continue

        for current, directories, filenames in os.walk(
            root, followlinks=False
        ):
            directories[:] = [
                name for name in directories
                if name not in {".git", "__pycache__"}
                and not name.startswith("LINEAGE_SCREEN_BATCH")
            ]

            for filename in filenames:
                path = Path(current) / filename
                if path.is_symlink():
                    continue

                kind = classify_asset(path)
                if kind in scan_kinds:
                    assets.append({
                        "kind": kind,
                        "path": str(path),
                        "member": "",
                    })
                elif path.suffix.lower() == ".zip":
                    archives.append(path)

    for path in archives:
        try:
            with zipfile.ZipFile(path) as archive:
                for member in archive.infolist():
                    if member.is_dir() or member.file_size > LIMIT:
                        continue
                    kind = classify_asset(member.filename)
                    if kind in scan_kinds:
                        assets.append({
                            "kind": kind,
                            "path": str(path),
                            "member": member.filename,
                        })
        except (OSError, zipfile.BadZipFile) as error:
            errors.append({
                "path": str(path),
                "error": repr(error),
            })

    pd.DataFrame(errors, columns=["path", "error"]).to_csv(
        out / "archive_read_errors.csv", index=False
    )

    unique = {}
    for asset in assets:
        unique[(asset["kind"],) + asset_id(asset)] = asset
    return list(unique.values())


def resolve_binary(kind, assets, audit):
    variants = collections.defaultdict(list)

    for asset in assets:
        if asset["kind"] != kind:
            continue
        try:
            digest = asset_hash(asset)
            variants[digest].append(dict(asset, sha256=digest))
            audit.append({
                **asset,
                "sha256": digest,
                "status": "READABLE",
            })
        except Exception as error:
            audit.append({
                **asset,
                "sha256": "",
                "status": repr(error),
            })

    if len(variants) != 1:
        raise InputProblem(
            f"{kind}: found {len(variants)} distinct readable contents. "
            "No arbitrary version will be selected."
        )

    return min(
        next(iter(variants.values())),
        key=lambda asset: (
            bool(asset.get("member")),
            len(asset["path"]),
            asset["path"],
            asset.get("member", ""),
        ),
    )


def resolve_clean(assets, selected, reconciliation, out):
    audit = []
    variants = collections.defaultdict(list)
    target_keys = set(selected.ks)
    target_labels = selected.set_index("ks").y.sort_index()
    required_keys = set(reconciliation.ks)

    require(
        reconciliation,
        ["ks", "clean_orig_pkts", "clean_orig_ip_bytes"],
        "Saved reconciliation",
    )
    reference = reconciliation.set_index("ks")

    for asset in assets:
        if asset["kind"] != "clean":
            continue

        record = dict(asset)
        try:
            frame = normalize(read_table(asset), labelled=True)
            if len(frame) != 1024 or set(frame.ks) != target_keys:
                raise InputProblem("Does not match the frozen target key set.")

            frame = frame.set_index("ks").loc[selected.ks].reset_index()

            labels = frame.set_index("ks").y.sort_index()
            if not labels.equals(target_labels):
                raise InputProblem("Labels differ from the frozen target.")

            if not required_keys.issubset(set(frame.ks)):
                raise InputProblem("Missing reconciliation keys.")

            indexed = frame.set_index("ks")
            pkts = numeric(reference.clean_orig_pkts)
            byte_counts = numeric(reference.clean_orig_ip_bytes)

            if not np.array_equal(
                indexed.loc[reference.index, "orig_pkts"].to_numpy(float),
                pkts.to_numpy(float),
            ):
                raise InputProblem("Clean packet counters differ.")

            if not np.array_equal(
                indexed.loc[
                    reference.index, "orig_ip_bytes"
                ].to_numpy(float),
                byte_counts.to_numpy(float),
            ):
                raise InputProblem("Clean byte counters differ.")

            benign = frame.y.eq(0)
            if not np.array_equal(
                frame.loc[benign, FEATURES[:-1]].to_numpy(float),
                selected.loc[benign, FEATURES[:-1]].to_numpy(float),
            ):
                raise InputProblem(
                    "Benign predictors differ although this pacing "
                    "operation targets selected malicious flows only."
                )

            if not np.array_equal(
                frame.loc[benign, "proto"].to_numpy(),
                selected.loc[benign, "proto"].to_numpy(),
            ):
                raise InputProblem("Benign protocols differ.")

            fingerprint = semantic_hash(frame)
            record.update({
                "status": "COMPATIBLE",
                "sha256": asset_hash(asset),
                "semantic_sha256": fingerprint,
            })
            variants[fingerprint].append((dict(record), frame))

        except Exception as error:
            record.update({
                "status": repr(error),
                "sha256": "",
                "semantic_sha256": "",
            })

        audit.append(record)

    pd.DataFrame(audit).to_csv(
        out / "clean_resolution.csv", index=False
    )

    if len(variants) != 1:
        readable = [
            row for row in audit if row["status"] == "COMPATIBLE"
        ]
        print("\nCLEAN INPUT RESOLUTION")
        print(pd.DataFrame(audit).to_string(index=False))
        raise InputProblem(
            f"Clean resolution found {len(variants)} distinct compatible "
            f"model-input tables from {len(readable)} compatible copies. "
            "Nothing was chosen using prediction accuracy."
        )

    copies = next(iter(variants.values()))
    asset, frame = min(
        copies,
        key=lambda pair: (
            bool(pair[0].get("member")),
            len(pair[0]["path"]),
            pair[0]["path"],
            pair[0].get("member", ""),
        ),
    )
    return asset, frame


def predict(model, frame):
    if list(model.classes_) != [0, 1]:
        raise InputProblem("Unexpected model class ordering.")
    probabilities = model.predict_proba(model_matrix(frame))[:, 1]
    if (
        not np.isfinite(probabilities).all()
        or (probabilities < 0).any()
        or (probabilities > 1).any()
    ):
        raise InputProblem("Invalid model probabilities.")
    return probabilities

            isinstance(child, ast.Call)
            and isinstance(child.func, ast.Attribute)
            and child.func.attr == "startswith"
            and isinstance(child.func.value, ast.Name)

def sha_text(text):
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def asset_hash(asset):
    digest = hashlib.sha256()
    if asset.get("member"):
        with zipfile.ZipFile(asset["path"]) as archive:
            with archive.open(asset["member"]) as handle:
                for block in iter(
                    lambda: handle.read(4 * 1024**2), b""
                ):
                    digest.update(block)
    else:
        with open(asset["path"], "rb") as handle:
            for block in iter(
                lambda: handle.read(4 * 1024**2), b""
            ):
                digest.update(block)
    return digest.hexdigest()

class ParameterizeDelay(ast.NodeTransformer):
    def __init__(self, delay):
        self.delay = delay
        self.delay_operations = 0
        self.asset_assignments = 0
        self.output_prefixes = 0

    def visit_BinOp(self, node):
        self.generic_visit(node)

        # Change only the delay multiplier 10 * 10**9.
        # The timestamp precision 10**9 remains unchanged.
        if (
            isinstance(node.op, ast.Mult)
            and isinstance(node.left, ast.Constant)
            and node.left.value == 10
            and isinstance(node.right, ast.BinOp)
            and isinstance(node.right.op, ast.Pow)
            and isinstance(node.right.left, ast.Constant)
            and node.right.left.value == 10
            and isinstance(node.right.right, ast.Constant)
            and node.right.right.value == 9
        ):
            node.left = ast.Constant(self.delay)
            self.delay_operations += 1
        return node

    def visit_Constant(self, node):
        if isinstance(node.value, str):
            value = node.value
            if "LINEAGE_TRANSFORM_BATCH3_" in value:
                value = value.replace(
                    "LINEAGE_TRANSFORM_BATCH3_",
                    f"LINEAGE_BATCH4_PACE_{self.delay}S_",
                )
                self.output_prefixes += 1

            # Update descriptive messages and the temporary PCAP name.
            # Does not change the model version string "8.0.10".
            value = value.replace("10s", f"{self.delay}s")
            node.value = value
        return node

    def visit_Dict(self, node):
        self.generic_visit(node)
        for index, key in enumerate(node.keys):
            if (
                isinstance(key, ast.Constant)
                and key.value == "delay_seconds"
                and isinstance(node.values[index], ast.Constant)
                and node.values[index].value == 10
            ):
                node.values[index] = ast.Constant(self.delay)
        return node

    def visit_Assign(self, node):
        self.generic_visit(node)
        if any(
            isinstance(target, ast.Name)
            and target.id == "assets"
            for target in node.targets
        ):
            self.asset_assignments += 1
            override = ast.parse(
                "assets['selected10'] = "
                "retarget_historical_input("
                f"assets['selected10'], {self.delay})"
            ).body[0]
            return [node, override]
        return node
